# HDRI Sampling

## Sampling the CDF Distributions

On the CPU side, we build the CDF distribution to importance sample from the HDRI equirectangular map when using NEE, which is then passed to the GPU side with a buffer:

```python
def _build_hdri_distribution(self):
    img = self.img.astype(np.float64)
    luminance = img[:, :, 0] * 0.2126 + img[:, :, 1] * 0.7152 + img[:, :, 2] * 0.0722

    # Adjust for equirectangular map spherical distortion
    theta = (np.arange(self.height) + 0.5) / self.height * np.pi
    sin_theta = np.sin(theta)[:, None]
    weighted = luminance * sin_theta

    # Build the CDFs
    # --------------
    # Marginal distribution over rows
    row_sums = weighted.sum(axis=1)
    row_cdf = np.cumsum(row_sums)
    row_cdf /= row_cdf[-1]

    # Conditional distribution over column given a row
    col_cdf = np.cumsum(weighted, axis=1)
    col_cdf /= col_cdf[:, -1:]

    self.row_cdf = row_cdf.astype(f4)
    self.col_cdf = col_cdf.astype(f4)
```

To sample from the the row and column CDFs in the shader,
we do binary searches to find the row and column cells from a uniform random number.
We also extract the high and lower bounds of the cell range and the PMF of choosing that cell.

Since the CDF distribution is defined with a linear slope between each cell, the PMF is the range of the CDF cell.

```glsl
int SampleRowCdf(float Xi, int height, out float rowHigh, out float rowLow, out float rowPmf) {
    // Binary search to find the correct row
    int lo = 0, hi = height - 1;
    while (lo < hi) {
        int mid = (hi + lo) / 2;
        float val = texelFetch(hdriRowCdf, ivec2(0, mid), 0).r;
        if (val < Xi) {
            lo = mid + 1;
        } else {
            hi = mid;
        }
    }
    // Use lo as the chosen row index
    rowHigh = texelFetch(hdriRowCdf, ivec2(0, lo), 0).r;
    rowLow = lo > 0 ? texelFetch(hdriRowCdf, ivec2(0, lo - 1), 0).r : 0.0;
    rowPmf = max(rowHigh - rowLow, 1e-8);
    return lo;
}
```

```glsl
int SampleColCdf(float Xi, int row, int width, out float colHigh, out float colLow, out float colPmf) {
    // Binary search to find the correct column
    int lo = 0, hi = width - 1;
    while (lo < hi) {
        int mid = (hi + lo) / 2;
        float val = texelFetch(hdriColCdf, ivec2(mid, row), 0).r;
        if (val < Xi) {
            lo = mid + 1;
        } else {
            hi = mid;
        }
    }
    colHigh = texelFetch(hdriColCdf, ivec2(lo, row), 0).r;
    colLow = lo > 0 ? texelFetch(hdriColCdf, ivec2(lo - 1, row), 0).r : 0.0;
    colPmf = max(colHigh - colLow, 1e-8);
    return lo;
}
```

## Directly Sampling the HDRI

Since sampling the CDF distributions only give us an integer position of the column and row,
it throws away the data of where inside that cell the uniformly generated value sits.
Thus, we calculate the required offset to map the chosen column and row back to a continuous position, then calculated the uv coordinates to sample the HDRI with:

```glsl
float dv = (Xi.x - rowLow) / rowPmf;
float du = (Xi.y - colLow) / colPmf;

float u = (float(col) + du) / float(width);
float v = (float(row) + dv) / float(height);
```

### HDRI PDF

To find the solid angle HDRI PDF $\text{PDF}_{\Omega}(\omega)$, we must first find the PDF of choosing the sampled continuous $(u, v)$ coordinates. We can find this PDF, which we call $\text{PDF}_{U,V}(u, v)$, with the following calculations:

$$
\begin{align*}
\text{PDF}_{U,V}(u, v) &= \dfrac{\text{PMF}(\text{row})\cdot\text{PMF}(\text{col | row})}{\Delta{u}\cdot\Delta{v}} \\
&= \text{PMF}(\text{row})\cdot\text{PMF}(\text{col | row})\cdot{W}\cdot{H}
\end{align*}
$$

where $W$ and $H$ represent the respective width and height of the HDRI texture.

To solve for $\text{PDF}_{\Omega}(\omega)$, we need to note that we need a PDF with respect to solid angle.
As such, we need to map $(u, v) \rightarrow (\theta, \phi) \rightarrow \omega$.
So,

$$
\text{PDF}_{\Omega}(\omega) =
\text{PDF}_{\Theta, \Phi}(\theta, \phi) \cdot |J_{\Omega}| =
\text{PDF}_{U, V}(u, v) \cdot |J_{\Theta, \Phi}| \cdot |J_{\Omega}|
$$

Since we define
$\theta = v\pi, \phi = (u-0.5)\cdot{2}\cdot\pi \implies v = \frac{\theta}{\pi}, u = \frac{\phi}{2\pi}+0.5$,

$$
J_{\Theta, \Phi} =
\begin{vmatrix}
\frac{\partial{u}}{\partial{\phi}} & \frac{\partial{u}}{\partial{\theta}} \\
\frac{\partial{v}}{\partial{\phi}} & \frac{\partial{v}}{\partial{\theta}}
\end{vmatrix} =
\begin{vmatrix}
\frac{1}{2\pi} & 0 \\
0 & \frac{1}{\pi}
\end{vmatrix} =
\dfrac{1}{2\pi} \cdot \dfrac{1}{\pi} - 0\cdot0 =
\dfrac{1}{2\pi^2}
$$

Also, the relationship between solid angle and spherical angles is
$d\omega = \sin{\theta}d\theta d\phi$. Because probabilistic mass is conserved,

$$
\text{PDF}_{\Theta, \Phi}(\theta, \phi)d\theta d\phi =
\text{PDF}_{\Omega}(\omega)d\omega =
\text{PDF}_{\Omega}(\omega)\sin{\theta}d\theta d\phi
$$

and so

$$
\text{PDF}_{\Omega}(\omega) = \dfrac{\text{PDF}_{\Theta, \Phi}(\theta, \phi)}{\sin{\theta}} \implies
J_{\Omega} = \dfrac{1}{\sin{\theta}}
$$

Finally, we get

$$
\text{PDF}_{\Omega}(\omega) = \text{PDF}_{U,V}(u,v) \cdot |J_{\Theta, \Phi}| \cdot |J_{\Omega}|
$$

For completeness,

$$
\text{PDF}_{\Omega}(\omega) = \text{PMF}(\text{row})\cdot\text{PMF}(\text{col | row})\cdot{W}\cdot{H} \cdot \left| \dfrac{1}{2\pi^2} \right| \cdot \left|\dfrac{1}{\sin{\theta}}\right|
$$

## Full Implementation

```glsl
vec3 DirectSampleHdri(inout uvec3 rng, out vec3 d, out float hdriPdf) {
    ivec2 size = textureSize(hdri, 0);
    int width = size.x, height = size.y;

    vec3 Xi = Pcg3d(rng);

    float rowHigh, rowLow, rowPmf;
    int row = SampleRowCdf(Xi.x, height, rowHigh, rowLow, rowPmf);
    float colHigh, colLow, colPmf;
    int col = SampleColCdf(Xi.y, row, width, colHigh, colLow, colPmf);

    // Offsets to the row and column column to map it back to a continuous point
    float dv = (Xi.x - rowLow) / rowPmf;
    float du = (Xi.y - colLow) / colPmf;

    // Sampled continuous point in ranges [0, 1)
    float u = (float(col) + du) / float(width);
    float v = (float(row) + dv) / float(height);

    float theta = v * PI;
    float phi = (u - 0.5) * 2.0 * PI;

    d = normalize(vec3(
        sin(theta) * cos(phi),
        cos(theta),
        sin(theta) * sin(phi)
    ));

    float mapPdf = rowPmf * colPmf * float(width) * float(height);
    float sinTheta = sin(theta);
    hdriPdf = sinTheta > 0.0 ? mapPdf / (2.0 * PI * PI * sinTheta) : 0.0;

    return texture(hdri, vec2(u, v)).rgb;
}
```

```glsl
float HdriPdf(vec3 d) {
    ivec2 size = textureSize(hdri, 0);
    int width = size.x, height = size.y;

    // Convert to spherical coordinates
    float phi = atan(d.z, d.x);
    float theta = acos(d.y);
    // Convert to uv coordinates
    vec2 uv = vec2(phi / (2.0 * PI) + 0.5, theta / PI);

    int col = min(int(uv.x * float(width)), width - 1);
    int row = min(int(uv.y * float(height)), height - 1);

    float rowHigh = texelFetch(hdriRowCdf, ivec2(0, row), 0).r;
    float rowLow = row > 0 ? texelFetch(hdriRowCdf, ivec2(0, row - 1), 0).r : 0.0;
    float rowPmf = max(rowHigh - rowLow, 1e-8);
    
    float colHigh = texelFetch(hdriColCdf, ivec2(col, row), 0).r;
    float colLow = col > 0 ? texelFetch(hdriColCdf, ivec2(col - 1, row), 0).r : 0.0;
    float colPmf = max(colHigh - colLow, 1e-8);

    float mapPdf = rowPmf * colPmf * float(width) * float(height);
    float sinTheta = sin(theta);
    return sinTheta > 0.0 ? mapPdf / (2.0 * PI * PI * sinTheta) : 0.0;
}
```
