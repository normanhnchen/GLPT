# 5.12 BSDF Evaluation

Note: this section is used for [light sampling](../7_Light_Sampling/7.1_Overview.md).

Since light sampling already gives the outgoing light direction, we skip the entirety of stochastic sampling. Thus, this gives us many functions used specifically for NEE and MIS.

## BRDF Evaluation

```glsl
vec3 EvaluateBrdf(vec3 wi, Ray ray, SurfaceInteraction si) {
    Material mat = si.mat;

    vec3 ns = si.ns;
    vec3 wo = -ray.d;

    vec3 F0 = vec3(0.04);
    F0 = mix(F0, mat.baseCol, mat.metallic);

    vec3 wh = normalize(wo + wi);

    float alpha = mat.roughness * mat.roughness;
    float alpha2 = alpha * alpha;

    float nsDotWh = dot(ns, wh);
    float nsDotWo = dot(ns, wo);
    float nsDotWi = dot(ns, wi);
    float whDotWo = dot(wh, wo);

    vec3 F = FresnelSchlick(max(whDotWo, 0.0), F0);
    float D = TrowbridgeReitzGgx(max(nsDotWh, 0.0), alpha);

    float G = SmithGgxMaskingShadowing(abs(nsDotWi), abs(nsDotWo), alpha2);

    vec3 diffuse = vec3(1.0) - F;
    diffuse *= 1.0 - mat.metallic;
    diffuse *= 1.0 - mat.transmission;

    vec3 specular = (D * G * F) / max(4.0 * max(nsDotWo, 0.0) * max(nsDotWi, 0.0), EPSILON);
    diffuse *= mat.baseCol / PI;

    vec3 brdf = (diffuse + specular) * max(nsDotWi, 0.0);

    return brdf;
}
```

```glsl
vec3 EvaluateBrdfAndPdf(vec3 wi, Ray ray, SurfaceInteraction si, out float brdfPdf) {
    Material mat = si.mat;

    vec3 ns = si.ns;
    vec3 wo = -ray.d;

    vec3 F0 = vec3(0.04);
    F0 = mix(F0, mat.baseCol, mat.metallic);

    vec3 wh = normalize(wo + wi);

    float alpha = mat.roughness * mat.roughness;
    float alpha2 = alpha * alpha;

    float nsDotWo = dot(ns, wo);
    float nsDotWi = dot(ns, wi);
    float nsDotWh = dot(ns, wh);
    float whDotWo = dot(wh, wo);

    vec3 F = FresnelSchlick(max(whDotWo, 0.0), F0);
    float D = TrowbridgeReitzGgx(max(nsDotWh, 0.0), alpha);

    float G1 = SmithGgxMasking(abs(nsDotWo), alpha2);
    float G2 = SmithGgxMaskingShadowing(abs(nsDotWi), abs(nsDotWo), alpha2);

    LobeProbs lobeProbs = ComputeLobeProbs(mat, max(nsDotWo, 0.0), F0);

    float specularPdf = (D * G1) / max((4.0 * max(nsDotWo, 0.0)), EPSILON) * lobeProbs.specular;
    float diffusePdf = CosineSampleHemispherePdf(max(nsDotWi, 0.0)) * lobeProbs.diffuse;
    brdfPdf = specularPdf + diffusePdf;

    vec3 diffuse = vec3(1.0) - F;
    diffuse *= 1.0 - mat.metallic;
    diffuse *= 1.0 - mat.transmission;

    vec3 specular = (D * G2 * F) / max(4.0 * max(nsDotWo, 0.0) * max(nsDotWi, 0.0), EPSILON);
    diffuse *= mat.baseCol / PI;

    vec3 brdf = (diffuse + specular) * max(nsDotWi, 0.0);

    return brdf;
}
```

## BTDF Evaluation

```glsl
vec3 EvaluateBtdf(vec3 wi, Ray ray, SurfaceInteraction si) {
    Material mat = si.mat;
    vec3 ns = si.ns;
    vec3 wo = -ray.d;
    float eta_i = si.eta_i;
    float eta_t = si.eta_t;

    float alpha = mat.roughness * mat.roughness;
    float alpha2 = alpha * alpha;

    vec3 wh = normalize(eta_i * wo + eta_t * wi);
    if (dot(ns, wh) < 0.0) wh = -wh;

    float nsDotWo = dot(ns, wo);
    float nsDotWi = dot(ns, wi);
    float nsDotWh = dot(ns, wh);
    float woDotWh = dot(wo, wh);
    float wiDotWh = dot(wi, wh);

    vec3 F0 = vec3(0.04);
    F0 = mix(F0, mat.baseCol, mat.metallic);
    vec3 F = FresnelSchlick(abs(woDotWh), F0);

    float D = TrowbridgeReitzGgx(max(nsDotWh, 0.0), alpha);

    float G2 = SmithGgxMaskingShadowing(abs(nsDotWi), abs(nsDotWo), alpha2);

    float denom = eta_i * woDotWh + eta_t * wiDotWh;
    denom = max(denom * denom, EPSILON);

    float jacobian = abs(woDotWh * wiDotWh) / max(abs(nsDotWo) * abs(nsDotWi), EPSILON);

    vec3 btdf = (vec3(1.0) - F) * D * G2 * jacobian * (eta_t * eta_t) / denom;

    return btdf;
}
```

```glsl
vec3 EvaluateBtdfAndPdf(vec3 wi, Ray ray, SurfaceInteraction si, out float btdfPdf) {
    Material mat = si.mat;
    vec3 ns = si.ns;
    vec3 wo = -ray.d;
    float eta_i = si.eta_i;
    float eta_t = si.eta_t;

    float alpha = mat.roughness * mat.roughness;
    float alpha2 = alpha * alpha;

    vec3 wh = normalize(eta_i * wo + eta_t * wi);
    if (dot(ns, wh) < 0.0) wh = -wh;

    float nsDotWo = dot(ns, wo);
    float nsDotWi = dot(ns, wi);
    float nsDotWh = dot(ns, wh);
    float woDotWh = dot(wo, wh);
    float wiDotWh = dot(wi, wh);

    vec3 F0 = vec3(0.04);
    F0 = mix(F0, mat.baseCol, mat.metallic);
    vec3 F = FresnelSchlick(abs(woDotWh), F0);

    float D = TrowbridgeReitzGgx(max(nsDotWh, 0.0), alpha);

    float G1 = SmithGgxMasking(abs(nsDotWo), alpha2);
    float G2 = SmithGgxMaskingShadowing(abs(nsDotWi), abs(nsDotWo), alpha2);

    float denom = eta_i * woDotWh + eta_t * wiDotWh;
    denom = max(denom * denom, EPSILON);

    LobeProbs lobeProbs = ComputeLobeProbs(mat, abs(nsDotWo), F0);

    float dwh_dwi = abs((eta_t * eta_t * wiDotWh) / max(denom, EPSILON));
    btdfPdf = D * G1 * abs(woDotWh) / max(abs(nsDotWo), EPSILON) * dwh_dwi;
    btdfPdf *= lobeProbs.transmission;

    float jacobian = abs(woDotWh * wiDotWh) / max(abs(nsDotWo) * abs(nsDotWi), EPSILON);

    vec3 btdf = (vec3(1.0) - F) * D * G2 * jacobian * (eta_t * eta_t) / denom;

    return btdf;
}
```

## BSDF Evaluation

```glsl
vec3 EvaluateBsdf(vec3 wi, Ray ray, SurfaceInteraction si) {
    vec3 ns = si.ns;
    float nsDotWi = dot(ns, wi);
    if (nsDotWi > 0.0) {
        /* Sample the BRDF */

        return EvaluateBrdf(wi, ray, si);
    } else {
        /* Sample the BTDF */

        return EvaluateBtdf(wi, ray, si);
    }
}
```

```glsl
vec3 EvaluateBsdfAndPdf(vec3 wi, Ray ray, SurfaceInteraction si, out float bsdfPdf) {
    float nsDotWi = dot(si.ns, wi);
    if (nsDotWi > 0.0) {
        /* Sample the BRDF */

        return EvaluateBrdfAndPdf(wi, ray, si, bsdfPdf);
    } else {
        /* Sample the BTDF */

        return EvaluateBtdfAndPdf(wi, ray, si, bsdfPdf);
    }
}
```
